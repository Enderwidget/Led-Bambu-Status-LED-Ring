function hsvToRgb(h,s,v){
  var c=v*s, x=c*(1-Math.abs((h/60)%2-1)), m=v-c, r,g,b;

function rgbToHsv(r,g,b){
  r/=255;g/=255;b/=255;
  var max=Math.max(r,g,b),min=Math.min(r,g,b),d=max-min,h;

}

void setup() {
  Serial.begin(115200);
  pinMode(RESET_BUTTON_PIN, INPUT_PULLUP);

  strip.begin();
  strip.setBrightness(255);
  setAllOff();

  loadPrinterConfig();
  loadColorConfig();

  wm.addParameter(&custom_ip);
  wm.addParameter(&custom_serial);
  wm.addParameter(&custom_code);
  wm.setSaveParamsCallback(saveParamsCallback);

  wm.setConfigPortalBlocking(false);
  wm.setConfigPortalTimeout(0);   // 0 = never close the setup network on its own

  bool connectedImmediately = wm.autoConnect("PrintLight-Setup");

  if (connectedImmediately) {
    Serial.println("Connected to saved WiFi immediately.");
    inConfigPortal = false;
    onWifiConnected();
  } else {
    Serial.println("Starting config portal (connect to 'PrintLight-Setup')...");
    appState = STATE_SETUP_PORTAL;
    inConfigPortal = true;
  }
}

// Runs once whenever we have a WiFi connection
void onWifiConnected() {
  appState = STATE_CONNECTING;

  Serial.print("WiFi connected. IP: ");
  Serial.println(WiFi.localIP());

  WiFi.setAutoReconnect(true);

  if (MDNS.begin("printlight")) {
    MDNS.addService("http", "tcp", 80);
    Serial.println("Settings page: http://printlight.local");
  }

  startSettingsAP();
  setupMQTT();
  setupWebServer();   // web server first, so the page works even if the printer is off
  connectMQTT();
}

// Runs our own WiFi network alongside the home WiFi, with a captive portal
void startSettingsAP() {
  if (apStarted) return;
  WiFi.mode(WIFI_AP_STA);
  if (strlen(AP_PASSWORD) >= 8) {
    WiFi.softAP(AP_NAME, AP_PASSWORD);
  } else {
    WiFi.softAP(AP_NAME);
  }
  delay(100);
  dnsServer.start(53, "*", WiFi.softAPIP());   // answer every DNS lookup with our IP
  apStarted = true;
  Serial.print("Settings network '");
  Serial.print(AP_NAME);
  Serial.print("' up at ");
  Serial.println(WiFi.softAPIP());
}

void setupMQTT() {
  secureClient.setInsecure(); // Bambu uses a self-signed certificate
  mqtt.setServer(printerIP, 8883);
  mqtt.setCallback(mqttCallback);
  mqtt.setBufferSize(8192);   // Increased buffer size for large Bambu MQTT payloads
}

void loop() {
  checkResetButton();

  if (inConfigPortal) {
    wm.process();

    if (WiFi.status() == WL_CONNECTED) {
      Serial.println("WiFi connected via portal!");
      inConfigPortal = false;
      onWifiConnected();
    } else if (wm.getWiFiIsSaved() && millis() > PORTAL_RETRY_MS) {
      // WiFi details exist but we couldn't join (router off, etc). Reboot and retry.
      Serial.println("Could not join saved WiFi, restarting to retry...");
      delay(200);
      ESP.restart();
    }
  } else {
    if (apStarted) dnsServer.processNextRequest();
    server.handleClient();          // serve the settings page first

    if (WiFi.status() == WL_CONNECTED) {
      if (!mqtt.connected()) {
        appState = STATE_CONNECTING;
        unsigned long now = millis();
        if (now - lastReconnectAttempt > 10000) {
          lastReconnectAttempt = now;
          connectMQTT();
        }
      } else {
        mqtt.loop();
      }
    } else {
      appState = STATE_CONNECTING;   // WiFi dropped; auto-reconnect is running
    }
  }

  updateAnimation();
}

void checkResetButton() {
  if (digitalRead(RESET_BUTTON_PIN) == LOW) {
    if (buttonHeldSince == 0) {
      buttonHeldSince = millis();
    } else if (!buttonHandled && millis() - buttonHeldSince > 3000) {
      buttonHandled = true;
      Serial.println("Reset button held - wiping settings and restarting...");
      setColor(strip.Color(255, 255, 255));
      delay(300);
      setAllOff();
      wm.resetSettings();
      prefs.begin("printer", false);
      prefs.clear();
      prefs.end();
      delay(500);
      ESP.restart();
    }
  } else {
    // Released: a short tap toggles the lights
    if (buttonHeldSince != 0 && !buttonHandled) {
      unsigned long held = millis() - buttonHeldSince;
      if (held > 50 && held < 3000) {
        ledsOn = !ledsOn;
        Serial.println(ledsOn ? "Lights ON" : "Lights OFF");
      }
    }
    buttonHeldSince = 0;
    buttonHandled = false;
  }
}

void connectMQTT() {
  if (WiFi.status() != WL_CONNECTED) return;
  if (strlen(printerIP) == 0 || strlen(printerSerial) == 0) return;

  // Quick reachability check so we don't sit in a long TLS timeout
  WiFiClient probe;
  if (!probe.connect(printerIP, 8883, 1500)) {
    Serial.println("Printer unreachable, will retry");
    return;
  }
  probe.stop();

  Serial.print("Connecting to printer MQTT...");
  String clientId = "printlight-" + String((uint32_t)ESP.getEfuseMac(), HEX);

  if (mqtt.connect(clientId.c_str(), "bblp", printerAccessCode)) {
    Serial.println("connected");
    String reportTopic = "device/" + String(printerSerial) + "/report";
    mqtt.subscribe(reportTopic.c_str());

    String requestTopic = "device/" + String(printerSerial) + "/request";
    String pushAllMsg = "{\"pushing\":{\"sequence_id\":\"0\",\"command\":\"pushall\"}}";
    mqtt.publish(requestTopic.c_str(), pushAllMsg.c_str());

    appState = STATE_IDLE;
  } else {
    Serial.print("failed, rc=");
    Serial.println(mqtt.state());
  }
}

void mqttCallback(char* topic, byte* payload, unsigned int length) {
  JsonDocument doc;
  DeserializationError err = deserializeJson(doc, payload, length);
  if (err) {
    Serial.print("JSON parse failed: ");
    Serial.println(err.c_str());
    return;
  }

  if (doc.containsKey("print")) {
    JsonObject printObj = doc["print"];

    if (printObj.containsKey("mc_percent")) {
      printPercent = printObj["mc_percent"].as<int>();
    }

    if (printObj.containsKey("gcode_state")) {
      String state = printObj["gcode_state"].as<String>();

      if (state == "PAUSE") {
        appState = STATE_PAUSED;
      } else if (state == "FAILED") {
        appState = STATE_ERROR;
      } else if (printPercent >= 100) {
        appState = STATE_FINISHED;
      } else if (printPercent > 0) {
        appState = STATE_PRINTING;
      } else {
        appState = STATE_IDLE;
      }
    } else if (printObj.containsKey("mc_percent")) {
      if (printPercent >= 100) {
        appState = STATE_FINISHED;
      } else if (printPercent > 0) {
        appState = STATE_PRINTING;
      }
    }
  }
}

void updateAnimation() {
  if (!ledsOn) {
    // Off: everything dark except LED 1, faint red so you know it's still powered
    for (int i = 0; i < LED_COUNT; i++) strip.setPixelColor(i, 0);
    strip.setPixelColor(0, strip.Color(STANDBY_RED, 0, 0));
    strip.show();
    return;
  }

  switch (appState) {
    case STATE_SETUP_PORTAL:
    case STATE_CONNECTING:
      chaseAnimation();
      break;

    case STATE_IDLE:
      setColor(scaleColor(colorIdle, brightIdle / 100.0));
      break;

    case STATE_PRINTING:
      progressFill(printPercent);
      break;

    case STATE_PAUSED:
      setColor(scaleColor(colorPaused, brightPaused / 100.0));
      break;

    case STATE_ERROR:
      setColor(scaleColor(colorError, brightError / 100.0));
      break;

    case STATE_FINISHED:
      setColor(scaleColor(colorPrintEnd, brightPrintEnd / 100.0));
      break;
  }
}

void chaseAnimation() {
  unsigned long now = millis();
  if (now - lastAnimStep < 60) return;
  lastAnimStep = now;

  uint32_t dimBlue = strip.Color(0, 0, 25);
  uint32_t brightBlue = strip.Color(60, 100, 255);

  for (int i = 0; i < LED_COUNT; i++) {
    int dist = (i - animIndex + LED_COUNT) % LED_COUNT;

    if (dist < 4) {
      float t = 1.0 - (dist / 4.0);
      strip.setPixelColor(i, lerpColor(dimBlue, brightBlue, t));
    } else {
      strip.setPixelColor(i, dimBlue);
    }
  }
  strip.show();

  animIndex = (animIndex + 1) % LED_COUNT;
}

void progressFill(int percent) {
  percent = constrain(percent, 0, 100);
  float litFloat = (percent / 100.0) * LED_COUNT;
  int litCount = (int)litFloat;
  if (litCount < 1) litCount = 1;

  float t = (float)percent / 100.0;
  float brightness = (brightPrintStart + (brightPrintEnd - brightPrintStart) * t) / 100.0;
  uint32_t baseColor = lerpColor(colorPrintStart, colorPrintEnd, t);
  uint32_t litColor = scaleColor(baseColor, brightness);

  for (int i = 0; i < LED_COUNT; i++) {
    if (i < litCount) {
      strip.setPixelColor(i, litColor);
    } else {
      strip.setPixelColor(i, strip.Color(0, 0, 0));
    }
  }
  strip.show();
}

uint32_t lerpColor(uint32_t colorA, uint32_t colorB, float t) {
  t = constrain(t, 0.0, 1.0);
  uint8_t rA = (colorA >> 16) & 0xFF, gA = (colorA >> 8) & 0xFF, bA = colorA & 0xFF;
  uint8_t rB = (colorB >> 16) & 0xFF, gB = (colorB >> 8) & 0xFF, bB = colorB & 0xFF;

  uint8_t r = rA + (rB - rA) * t;
  uint8_t g = gA + (gB - gA) * t;
  uint8_t b = bA + (bB - bA) * t;

  return strip.Color(r, g, b);
}

uint32_t scaleColor(uint32_t color, float factor) {
  factor = constrain(factor, 0.0, 1.0);
  uint8_t r = (color >> 16) & 0xFF, g = (color >> 8) & 0xFF, b = color & 0xFF;

  int rOut = (int)(r * factor);
  int gOut = (int)(g * factor);
  int bOut = (int)(b * factor);

  if (rOut > MAX_CHANNEL_VALUE) rOut = MAX_CHANNEL_VALUE;
  if (gOut > MAX_CHANNEL_VALUE) gOut = MAX_CHANNEL_VALUE;
  if (bOut > MAX_CHANNEL_VALUE) bOut = MAX_CHANNEL_VALUE;

  return strip.Color(rOut, gOut, bOut);
}

void setColor(uint32_t color) {
  for (int i = 0; i < LED_COUNT; i++) {
    strip.setPixelColor(i, color);
  }
  strip.show();
}

void setAllOff() {
  setColor(strip.Color(0, 0, 0));
}

String colorToHex(uint32_t color) {
  char buf[8];
  snprintf(buf, sizeof(buf), "#%06X", color);
  return String(buf);
}

uint32_t hexToColor(String hex) {
  hex.replace("#", "");
  return (uint32_t) strtol(hex.c_str(), NULL, 16);
}

void setupWebServer() {
  if (webServerStarted) return;
  webServerStarted = true;

  server.on("/", handleRoot);
  server.on("/save", HTTP_POST, handleSave);
  server.on("/listpresets", HTTP_GET, handleListPresets);
  server.on("/loadpreset", HTTP_GET, handleLoadPreset);
  server.on("/savepreset", HTTP_POST, handleSavePreset);
  server.on("/deletepreset", HTTP_GET, handleDeletePreset);
  server.onNotFound(handleNotFound);
  server.begin();
  Serial.print("Settings page ready at http://");
  Serial.println(WiFi.localIP());
}

// Captive portal: phones probe a known URL; redirecting them makes the sign-in popup appear
void handleNotFound() {
  String host = server.hostHeader();
  bool ours = (host == WiFi.softAPIP().toString() ||
               host == WiFi.localIP().toString() ||
               host == "printlight.local" || host.length() == 0);
  if (ours) {
    handleRoot();
    return;
  }
  server.sendHeader("Location", String("http://") + WiFi.softAPIP().toString() + "/", true);
  server.send(302, "text/plain", "");
}

void handleRoot() {
  // Send in chunks so we never hold the whole page in RAM as one String
  server.setContentLength(CONTENT_LENGTH_UNKNOWN);
  server.send(200, "text/html", "");

  server.sendContent_P(HTML_HEAD);

  String part = "<label>Printer IP</label><input type='text' id='printerIP' value='" + String(printerIP) + "'>";
  part += "<label>Printer Serial</label><input type='text' id='printerSerial' value='" + String(printerSerial) + "'>";
  part += "<label>Access Code</label><input type='text' id='printerCode' value='" + String(printerAccessCode) + "'>";
  server.sendContent(part);

  server.sendContent_P(HTML_BODY);

  String states = "var states = {";
  states += "idle:{color:'" + colorToHex(colorIdle) + "',bright:" + String(brightIdle) + "},";
  states += "paused:{color:'" + colorToHex(colorPaused) + "',bright:" + String(brightPaused) + "},";
  states += "error:{color:'" + colorToHex(colorError) + "',bright:" + String(brightError) + "},";
  states += "pstart:{color:'" + colorToHex(colorPrintStart) + "',bright:" + String(brightPrintStart) + "},";
  states += "pend:{color:'" + colorToHex(colorPrintEnd) + "',bright:" + String(brightPrintEnd) + "}";
  states += "};";
  server.sendContent(states);

  server.sendContent_P(HTML_SCRIPT);

  server.sendContent("");   // terminate chunked response
}

void handleSave() {
  bool printerChanged = false;

  if (server.hasArg("printerIP") && server.arg("printerIP") != String(printerIP)) {
    strncpy(printerIP, server.arg("printerIP").c_str(), sizeof(printerIP) - 1);
    printerIP[sizeof(printerIP) - 1] = '\0';
    printerChanged = true;
  }
  if (server.hasArg("printerSerial") && server.arg("printerSerial") != String(printerSerial)) {
    strncpy(printerSerial, server.arg("printerSerial").c_str(), sizeof(printerSerial) - 1);
    printerSerial[sizeof(printerSerial) - 1] = '\0';
    printerChanged = true;
  }
  if (server.hasArg("printerCode") && server.arg("printerCode") != String(printerAccessCode)) {
    strncpy(printerAccessCode, server.arg("printerCode").c_str(), sizeof(printerAccessCode) - 1);
    printerAccessCode[sizeof(printerAccessCode) - 1] = '\0';
    printerChanged = true;
  }
  if (printerChanged) {
    savePrinterConfig();
  }

  if (server.hasArg("idle"))   colorIdle       = hexToColor(server.arg("idle"));
  if (server.hasArg("paused")) colorPaused     = hexToColor(server.arg("paused"));
  if (server.hasArg("error"))  colorError      = hexToColor(server.arg("error"));
  if (server.hasArg("pstart")) colorPrintStart = hexToColor(server.arg("pstart"));
  if (server.hasArg("pend"))   colorPrintEnd   = hexToColor(server.arg("pend"));

  if (server.hasArg("idleB"))   brightIdle       = server.arg("idleB").toInt();
  if (server.hasArg("pausedB")) brightPaused     = server.arg("pausedB").toInt();
  if (server.hasArg("errorB"))  brightError      = server.arg("errorB").toInt();
  if (server.hasArg("pstartB")) brightPrintStart = server.arg("pstartB").toInt();
  if (server.hasArg("pendB"))   brightPrintEnd   = server.arg("pendB").toInt();

  saveColorConfig();
  server.send(200, "text/plain", "OK");

  if (printerChanged) {
    mqtt.disconnect();
    setupMQTT();
    connectMQTT();
  }
}

void handleListPresets() {
  String json = loadPresetsJson();
  JsonDocument doc;
  deserializeJson(doc, json);

  JsonDocument outDoc;
  JsonArray outArr = outDoc.to<JsonArray>();
  for (JsonObject p : doc.as<JsonArray>()) {
    outArr.add(p["name"].as<String>());
  }
  String out;
  serializeJson(outArr, out);
  server.send(200, "application/json", out);
}

void handleLoadPreset() {
  String name = server.arg("name");
  String json = loadPresetsJson();
  JsonDocument doc;
  deserializeJson(doc, json);

  for (JsonObject p : doc.as<JsonArray>()) {
    if (p["name"].as<String>() == name) {
      String out;
      serializeJson(p, out);
      server.send(200, "application/json", out);
      return;
    }
  }
  server.send(404, "text/plain", "Not found");
}

void handleSavePreset() {
  String name = server.arg("name");
  if (name.length() == 0) {
    server.send(400, "text/plain", "Missing name");
    return;
  }

  String json = loadPresetsJson();
  JsonDocument doc;
  deserializeJson(doc, json);
  JsonArray arr = doc.as<JsonArray>();

  for (int i = arr.size() - 1; i >= 0; i--) {
    if (arr[i]["name"].as<String>() == name) {
      arr.remove(i);
    }
  }

  while (arr.size() >= MAX_PRESETS) {
    arr.remove(0);
  }

  JsonObject newPreset = arr.add<JsonObject>();
  newPreset["name"]    = name;
  newPreset["idle"]    = server.arg("idle");
  newPreset["idleB"]   = server.arg("idleB").toInt();
  newPreset["paused"]  = server.arg("paused");
  newPreset["pausedB"] = server.arg("pausedB").toInt();
  newPreset["error"]   = server.arg("error");
  newPreset["errorB"]  = server.arg("errorB").toInt();
  newPreset["pstart"]  = server.arg("pstart");
  newPreset["pstartB"] = server.arg("pstartB").toInt();
  newPreset["pend"]    = server.arg("pend");
  newPreset["pendB"]   = server.arg("pendB").toInt();

  String out;
  serializeJson(arr, out);
  savePresetsJson(out);

  server.send(200, "text/plain", "OK");
}

void handleDeletePreset() {
  String name = server.arg("name");
  String json = loadPresetsJson();
  JsonDocument doc;
  deserializeJson(doc, json);
  JsonArray arr = doc.as<JsonArray>();

  for (int i = arr.size() - 1; i >= 0; i--) {

    }
  }

  server.send(200, "text/plain", "OK");
}
